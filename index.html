            <h3>

                <span>📋 실시간 선택 현황</span>

            </h3>

            <ul class="result-list" id="history">

                <li style="color:#a0aec0; justify-content:center;">아직 선택한 인원이 없습니다.</li>

        <button class="admin-btn" onclick="resetGame()">진행자 전용 판 리셋하기 🔄</button>

    <script>

        var outcomes = ['🎉 당첨!', '🎉 당첨!', '🎉 당첨!', '💥 꽝'];

        var historyData = [];

        var openedCards = [false, false, false, false];


        function shuffleOutcomes() {

            for (var i = outcomes.length - 1; i > 0; i--) {

                var j = Math.floor(Math.random() * (i + 1));

                var temp = outcomes[i];

                outcomes[i] = outcomes[j];

                outcomes[j] = temp;

            }

        }


        function resetGame() {

            var password = prompt('진행자 비밀번호를 입력하세요:');

            if (password !== 'boram1234') {

                if (password !== null) alert('비밀번호가 틀렸습니다!');

                return;

            }


            shuffleOutcomes();

            historyData = [];

            openedCards = [false, false, false, false];

            document.getElementById('username').value = '';

            document.getElementById('history').innerHTML = '<li style="color:#a0aec0; justify-content:center;">아직 선택한 인원이 없습니다.</li>';


            for (var i = 0; i < 4; i++) {

                var cardEl = document.getElementById('card-' + i);

                cardEl.className = 'card';

                cardEl.innerHTML = (i + 1) + '번 카드';

            }

            alert('판이 리셋되었습니다!');

        }


        function pickCard(index) {

            var nameInput = document.getElementById('username');

            var nickname = nameInput.value.trim();


            if (!nickname) {

                alert('카드 뽑기 전 닉네임을 입력해주세용!');

                nameInput.focus();

                return;

            }


            if (openedCards[index]) {

                alert('이미 열린 카드입니다!');

                return;

            }


            if (!confirm((index + 1) + '번 카드를 뽑으시겠습니까?')) return;


            openedCards[index] = true;

            var result = outcomes[index];

            var cardEl = document.getElementById('card-' + index);

            var isWin = result.indexOf('당첨') !== -1;


            cardEl.className = 'card opened ' + (isWin ? 'win' : 'lose');

            cardEl.innerHTML = (isWin ? '🎉<br>당첨' : '💥<br>꽝!') + '<div class="user-tag">' + nickname + '</div>';


            if (historyData.length === 0) {

                document.getElementById('history').innerHTML = '';

            }


            var badgeClass = isWin ? 'badge-win' : 'badge-lose';

            var recordHtml = '<li><span><b>' + nickname + '</b> (' + (index + 1) + '번)</span><span class="' + badgeClass + '">' + result + '</span></li>';

            

            document.getElementById('history').insertAdjacentHTML('afterbegin', recordHtml);

            historyData.push({ nickname: nickname, card: index + 1, result: result });

            nameInput.value = "";

        }


        shuffleOutcomes();

    </script>
